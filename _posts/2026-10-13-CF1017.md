### B.

已知最终的区间 $$[l, r]$$ ，这是n天之后的结果，求解从开始到第m天的结果

注意 $l \leq 0 \leq r$ ，可以进行逆向推导，也就是通过最终结果往前推导 n - m 天

但是需要注意不要越界，$l = 0 | r= 0$ 时,则停止当前边的遍历，转向另一边

```c++
#include <bits/stdc++.h>

using namespace std;
using i64 = long long;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    
    int t; cin >> t;
    while (t--) {
        int n, m, l, r; cin >> n >> m >> l >> r;
        int cost = n - m;
        
        while (cost) {
            if (l == 0) break;
            l++;
            cost--;
        }
        
        if (cost) {
            r -= cost;
        }
        
        cout << l << ' ' << r << '\n';
    }
    
    return 0;
}
```

### C.

得到一个 n * n 的二维数组，保证1个长度为2*n的排列一定存在

目前可以知道这个排列就是第一列+最后一行或者第一行+最后一列（去除重复元素）

缺少的那个元素是排列中存在，但遍历不存在。

解法：先遍历第一行和最后一列，遍历到的数字push_back()的一个output数组后面

遍历完成后，找出那个缺少的元素，优先输出这个缺少的元素，然后逐个输出output中的元素



```c++
#include <bits/stdc++.h>

using namespace std;
using i64 = long long;

int main() {
    ios::sync_with_stdio(0);
    cin.tie(0);
    
    int t; cin >> t;
    while (t--) {
		int n; cin >> n;
        vector<vector<int>> g(n, vector<int>(n));
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                cin >> g[i][j];
            }
        }
        
        vector<int> output;
        vector<int> a(2000, 0);
        
        // 遍历第一行
        for (int i = 0; i < n; i++) {
            output.push_back(g[0][i]);
            a[g[0][i]]++;
        }
        
        // 遍历最后一列
        for (int i = 1; i < n; i++) {
            output.push_back(g[i][n-1]);
            a[g[i][n-1]]++;
        }
        
        for (int i = 1; i <= 2 * n; i++) {
            if (a[i] == 0) {
                cout << i << ' ';
                break;
            }
        }
        
        for (int i = 0; i < output.size(); i++) {
            cout << output[i];
            if (i != output.size() - 1) {
                cout << ' ';
            } else {
                cout << '\n';
            }
        }
    }
    
    return 0;
}
```
